  if (!_db) return { ok:false, reason:'no-db' };

  for (const k of Object.keys(_VISMA_NO_FIELDS)) {
    const { data, error } = await _db.from('romtegner_projects').select('id, name, project_no, org_id, deleted_at, ' + _VISMA_NO_COLS).not(_VISMA_NO_FIELDS[k].col, 'is', null);

  const seks = v => !v || (String(v).length === VISMA_NO_DIGITS && /^[0-9]+$/.test(String(v)));   // strengt: ingen rens
  const linje = r => ({ prosjektnr: _fmtProjectNo(r.project_no), navn: r.name, bedrift: (org(r.org_id) || {}).name || r.org_id || '(ingen)',
    tilbudsnr: r.visma_offer_no, ordrenr: r.visma_order_no, papirkurv: !!r.deleted_at });
  const alle = [...byId.values()].filter(r => r.visma_offer_no || r.visma_order_no);
  const ikke6 = alle.filter(r => !seks(r.visma_offer_no) || !seks(r.visma_order_no));
  const uten = alle.filter(r => { const o = org(r.org_id); return !!o && !String(o.erp_format || '').trim(); });
  const ukjent = alle.filter(r => !org(r.org_id)).length;
  const ut = { ok:true, medNummer: alle.length, ikke6siffer: ikke6.map(linje), utenVisma: uten.map(linje), ukjentBedrift: ukjent };

  if (ikke6.length) console.table(ut.ikke6siffer);
  if (uten.length) console.table(ut.utenVisma);
  return ut;
}
