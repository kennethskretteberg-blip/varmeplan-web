  const box = document.getElementById('wf-k-visma');
  let block = false;
  if (box && box.dataset.loaded === '1') for (const k of Object.keys(_VISMA_NO_FIELDS)) {
    const el = document.getElementById('wf-k-visma-' + k); if (!el) continue;
    const st = _vismaNoFieldState(el.value, el.dataset.orig || '');
    const msg = document.getElementById('wf-k-visma-msg-' + k); if (msg) msg.style.display = st.valid ? 'none' : '';
    if (st.blocks && !el.disabled) block = true;
  }
  const btn = document.getElementById('wf-k-save');
  if (btn) { btn.disabled = block; btn.style.opacity = block ? '.5' : ''; btn.style.cursor = block ? 'not-allowed' : ''; }
  return !block;
}
