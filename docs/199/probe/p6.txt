// Leverandør: hvert valg står rett under statusen det hører til. Installatør: foran Papirkurv.
const _PL_STATUS_OPTS_VISMA = [['uten_tilbudsnr','Tilbud sendt uten tilbudsnr.','tilbud_sendt'],['uten_ordrenr','Vunnet uten ordrenr.','vunnet']];
function _plStatusOpts(mode, visma) {
  const base = mode === 'stages' ? _PL_STATUS_OPTS_STAGES : _PL_STATUS_OPTS_SUPPLIER;
  if (!visma) return base;
  const out = [];
  for (const o of base) {
    if (mode === 'stages' && o[0] === 'trash') _PL_STATUS_OPTS_VISMA.forEach(x => out.push([x[0], x[1]]));
    out.push(o);
    if (mode !== 'stages') _PL_STATUS_OPTS_VISMA.filter(x => x[2] === o[0]).forEach(x => out.push([x[0], x[1]]));
  }
  return out;
}
function _plSyncStatusSelect() {
  const sel = document.getElementById('dash-filter-status');
  if (!sel) return;
  const mode = _plUseStages() ? 'stages' : 'supplier';
  const visma = (_vismaNoOrgUsesVisma() && !_vismaNoDisabled) ? '1' : '';
  if (sel.dataset.mode === mode && (sel.dataset.visma || '') === visma) return;
  const sameMode = sel.dataset.mode === mode, prev = sel.value;
  const opts = _plStatusOpts(mode, !!visma);
  sel.innerHTML = opts.map(([v, l]) => `<option value="${v}">${l}</option>`).join('');
  // Byttet visningen (leverandør ↔ installatør): standardvalget, som før. Kom eller forsvant bare
  // Visma-valgene (bedriften ble lastet, kolonnene manglet): behold det brukeren står på hvis det finnes.
  sel.value = (sameMode && opts.some(o => o[0] === prev)) ? prev : (mode === 'stages' ? 'stage' : 'active');
  sel.dataset.mode = mode;
  sel.dataset.visma = visma;
}
